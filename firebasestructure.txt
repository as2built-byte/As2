on a une base de donnee firebase avec AUTHENTICATION email et mot de passe active et les collections suivantes
users
  └── {uid}
      - role: "enterprise" | "expert" | "admin"
      - email
      - firstName
      - lastName
      - phone
      - createdAt
      - status: "active" |"pending"| "inactive"
experts
  └── {uid}
      - certifications: [] // vide par defaut rempli apres completer des formation
      - cvUrl
      - availability //true par defaut 
      - createdAt
enterprises
  └── {uid}
      - companyName
      - certifications: []
      - projectCount: number // 0 par defaut, incremente a chaque projet cree
      - hasSubscription: boolean // false par defaut, true apres approbation admin
      - subscriptionRequestPending: boolean // false par defaut, true apres demande d'abonnement, remis a false apres approbation
      - createdAt

projects
  └── {projectId}
      - enterpriseId: string // uid de l'entreprise proprietaire
      - title: string
      - description: string
      - address: string
      - startDate: Timestamp
      - status: "active" | "completed" | "archived"
      - createdAt: Timestamp
      - updatedAt: Timestamp

missions
  └── {missionId}
      - projectId: string // reference au projet
      - enterpriseId: string // uid de l'entreprise
      - expertId: string | null // null jusqu'a affectation par admin
      - title: string
      - description: string
      - status: "pending_admin" | "proposed" | "accepted" | "refused" | "completed"
          // pending_admin: en attente d'affectation par admin
          // proposed: propose a un expert, en attente de reponse
          // accepted: expert a accepte, mission en cours
          // refused: expert a refuse, retourne a pending_admin
          // completed: mission terminee
      - createdAt: Timestamp
      - updatedAt: Timestamp

formations
  └── {formationId}
      - title: "BIM Architecture – Fondamentaux"
      - description: "Apprendre les bases du BIM et Revit"
      - durationHours: 14
      - price: 40000 // en DZD
      - isActive: true
      - coverUrl: "" //le user aploader cet image selon un bon controle 
      - createdAt
      - updatedAt

packs
  └── {packId}
      - title: "Pack BIM Complet"
      - formationIds: []
      - price: 90000 // en DZD
      - createdAt
      - updatedAt
      // Note: isActive est calculé côté client - un pack est inactif si une de ses formations est inactive

payments
  └── {paymentId}
      - userId
      - itemType: "formation" | "pack" | "audit"
      - itemId
      - createdAt

notifications
  └── {notificationId}
      - type: "new_payment" | "new_registration" | "certification_request" | "subscription_request" | "new_mission"
      - title: string
      - message: string
      - data: {
          userId: string
          userName: string
          userRole: "expert" | "enterprise"
          itemType: "formation" | "pack" | "project" | "mission"  // selon le type
          itemId: string
          itemTitle: string
          amount: number // optionnel, pour les paiements
        }
      - targetRole: "admin"
      - read: boolean // false par défaut
      - createdAt // Timestamp Firebase
